// Supabase Edge Function : add-passage // Appelée depuis l'interface boutique quand l'employé scanne un client et clique "+1 passage". // - Incrémente les compteurs du client // - Génère une remise si un palier est atteint (6 -> 10%, 10 -> 15%, puis le cycle repart) // - Met à jour la carte Google Wallet du client en temps réel (le solde affiché change instantanément) // // Déploiement : coller ce code dans une nouvelle fonction Supabase nommée "add-passage" // (mêmes secrets que hyper-api : GOOGLE_SERVICE_ACCOUNT_JSON déjà configuré, réutilisé ici) import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; import { create, Header, Payload } from "https://deno.land/x/djwt@v2.8/mod.ts"; const ISSUER_ID = "BCR2DN6DTKFKJWQ3"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { const { client_id, boutique_id, employe } = await req.json(); if (!client_id || !boutique_id) { return json({ error: "client_id et boutique_id sont requis" }, 400); } const supabase = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); // 0. Anti-abus : un client ne peut gagner qu'un point toutes les 10 minutes // (empêche de scanner plusieurs fois d'affilée le QR de la boutique sans achat réel) const { data: dernierPassage } = await supabase .from("historique_passages") .select("date_passage") .eq("client_id", client_id) .order("date_passage", { ascending: false }) .limit(1) .single(); if (dernierPassage) { const minutesDepuisDernier = (Date.now() - new Date(dernierPassage.date_passage).getTime()) / 60_000; if (minutesDepuisDernier < 10) { const minutesRestantes = Math.ceil(10 - minutesDepuisDernier); return json( { error: "trop_tot", message: `Vous avez déjà gagné un point récemment. Revenez dans ${minutesRestantes} min.`, }, 429 ); } } // 1. Récupérer le client actuel const { data: client, error: clientError } = await supabase .from("clients_fidelite") .select("*") .eq("id", client_id) .single(); if (clientError || !client) { return json({ error: "Client introuvable. Vérifiez le QR code scanné." }, 404); } // 2. Enregistrer le passage dans l'historique await supabase.from("historique_passages").insert({ client_id: client.id, boutique_id: boutique_id, employe: employe || null, }); // 3. Incrémenter les compteurs let nouveauCycle = client.nb_passages_cycle_actuel + 1; const nouveauTotal = client.nb_passages_total + 1; let remiseGeneree: { type: string; pourcentage: number; code_promo: string } | null = null; // 4. Vérifier les paliers (6 -> 10%, 10 -> 15% puis le cycle repart à 0) if (nouveauCycle === 6) { const code = genererCodePromo("FIDELE10"); await supabase.from("remises").insert({ client_id: client.id, type: "fidelite", pourcentage: 10, code_promo: code, date_expiration: dansNJours(30), statut: "active", }); remiseGeneree = { type: "palier_6", pourcentage: 10, code_promo: code }; } else if (nouveauCycle >= 10) { const code = genererCodePromo("FIDELE15"); await supabase.from("remises").insert({ client_id: client.id, type: "fidelite", pourcentage: 15, code_promo: code, date_expiration: dansNJours(30), statut: "active", }); remiseGeneree = { type: "palier_10", pourcentage: 15, code_promo: code }; nouveauCycle = 0; // le cycle repart pour la prochaine série de passages } // 5. Mettre à jour le client en base const { data: updatedClient, error: updateError } = await supabase .from("clients_fidelite") .update({ nb_passages_cycle_actuel: nouveauCycle, nb_passages_total: nouveauTotal, }) .eq("id", client.id) .select() .single(); if (updateError) throw updateError; // 6. Mettre à jour la carte Google Wallet du client (en temps réel) let walletUpdated = false; try { await updateWalletObject(updatedClient, boutique_id, supabase); walletUpdated = true; } catch (walletErr) { // On ne bloque pas la réponse si la mise à jour Wallet échoue : // le passage est bien enregistré en base, seul l'affichage sur le tel du client peut avoir un délai. console.error("Erreur mise à jour Wallet:", walletErr); } return json({ success: true, client: { prenom: updatedClient.prenom, nb_passages_cycle_actuel: updatedClient.nb_passages_cycle_actuel, nb_passages_total: updatedClient.nb_passages_total, }, remise: remiseGeneree, walletUpdated, }); } catch (err) { return json({ error: String(err) }, 500); } }); function json(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } function genererCodePromo(prefix: string): string { const suffix = Math.random().toString(36).substring(2, 6).toUpperCase(); return `${prefix}-${suffix}`; } function dansNJours(n: number): string { const d = new Date(); d.setDate(d.getDate() + n); return d.toISOString(); } async function updateWalletObject(client: any, boutiqueId: string, supabase: any) { const { data: boutique } = await supabase .from("boutiques") .select("nom") .eq("id", boutiqueId) .single(); if (!boutique) return; const classSuffix = "pilvite_" + boutique.nom .toLowerCase() .replace("pil'vite ", "") .normalize("NFD").replace(/[\u0300-\u036f]/g, "") .replace(/[^a-z0-9]+/g, "_") .replace(/^_|_$/g, ""); const classId = `${ISSUER_ID}.${classSuffix}`; const objectId = `${ISSUER_ID}.client_${client.id}`; const serviceAccount = JSON.parse(Deno.env.get("GOOGLE_SERVICE_ACCOUNT_JSON")!); const accessToken = await getGoogleAccessToken(serviceAccount); const patchBody = { loyaltyPoints: { label: "Points de fidélité", balance: { string: `${client.nb_passages_cycle_actuel} / 6` }, }, }; await fetch( `https://walletobjects.googleapis.com/walletobjects/v1/loyaltyObject/${objectId}`, { method: "PATCH", headers: { Authorization: `Bearer ${accessToken}`, "Content-Type": "application/json", }, body: JSON.stringify(patchBody), } ); } async function getGoogleAccessToken(serviceAccount: any): Promise { const header: Header = { alg: "RS256", typ: "JWT" }; const now = Math.floor(Date.now() / 1000); const payload: Payload = { iss: serviceAccount.client_email, scope: "https://www.googleapis.com/auth/wallet_object.issuer", aud: "https://oauth2.googleapis.com/token", iat: now, exp: now + 3600, }; const privateKey = await importPrivateKey(serviceAccount.private_key); const assertion = await create(header, payload, privateKey); const res = await fetch("https://oauth2.googleapis.com/token", { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: new URLSearchParams({ grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer", assertion, }), }); const data = await res.json(); return data.access_token; } async function importPrivateKey(pem: string): Promise { const pemContents = pem .replace("-----BEGIN PRIVATE KEY-----", "") .replace("-----END PRIVATE KEY-----", "") .replace(/\s/g, ""); const binaryDer = Uint8Array.from(atob(pemContents), (c) => c.charCodeAt(0)); return crypto.subtle.importKey( "pkcs8", binaryDer, { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" }, false, ["sign"] ); }
